<script>
(function(){
  "use strict";
  var hist=[], busy=false;

  var SYS = [
    "You are an expert academic research assistant specialized in finding open Calls for Papers (CFPs).",
    "Your primary task is to search the web and find currently open or upcoming CFPs, special issues,",
    "conference submissions, and journal submission opportunities.",
    "",
    "The user is part of PAIC Econometrics at FAE Business School, researching:",
    "- GAMLSS and MSGARCH models for volatility and distributional analysis of agricultural commodity returns",
    "- Multi-objective portfolio optimization using evolutionary algorithms (NSGA-II, DEoptim)",
    "- Reinforcement Learning (Q-Learning, K-Bandit) for dynamic multi-period portfolio allocation",
    "- Agricultural commodity markets (corn, soybean, wheat futures)",
    "- Financial econometrics, regime switching, Bayesian methods",
    "",
    "For each CFP found, provide: **Name**, **Type** (journal/conference/workshop/special issue),",
    "**Deadline**, **URL**, and a brief description. Search IEEE (CEC, WCCI, SSCI), ACM, INFORMS,",
    "Elsevier (EJOR, Expert Systems), Springer, MDPI, Wiley, Taylor & Francis, and specialized venues.",
    "Prioritize opportunities with upcoming deadlines in 2025-2026. Always include URLs.",
    "Respond in English."
  ].join("\n");

  function $(id){return document.getElementById(id)}

  window.cfpCheckKey=function(){
    var k=$("gKey").value.trim();
    $("kDot").className="cfp-key-dot "+(k.length>10?"on":"off");
    $("cfpSendBtn").disabled=k.length<=10;
  };

  function addMsg(role,html,sources){
    var w=$("cfpWelcome"); if(w) w.style.display="none";
    var area=$("cfpChat"), d=document.createElement("div");
    d.className="cfp-msg "+role;
    var av='<div class="cfp-avatar">'+(role==="user"?"&#128100;":"&#129302;")+'</div>';
    var src="";
    if(sources&&sources.length){
      src='<div class="cfp-sources">';
      sources.forEach(function(s){
        var dom=""; try{dom=new URL(s.uri).hostname.replace("www.","")}catch(e){dom=s.uri}
        src+='<a class="cfp-src-chip" href="'+s.uri+'" target="_blank" rel="noopener">&#128279; '+(s.title||dom)+'</a>';
      });
      src+='</div>';
    }
    d.innerHTML=av+'<div class="cfp-bubble">'+html+src+'</div>';
    area.appendChild(d); area.scrollTop=area.scrollHeight;
  }

  function addTyping(){
    var d=document.createElement("div"); d.className="cfp-msg bot"; d.id="cfpTyping";
    d.innerHTML='<div class="cfp-avatar">&#129302;</div><div class="cfp-bubble"><div class="cfp-typing"><span></span><span></span><span></span></div></div>';
    $("cfpChat").appendChild(d); $("cfpChat").scrollTop=$("cfpChat").scrollHeight;
  }

  function renderMd(t){
    return t
      .replace(/\*\*(.+?)\*\*/g,'<strong>$1</strong>')
      .replace(/\*(.+?)\*/g,'<em>$1</em>')
      .replace(/`([^`]+)`/g,'<code>$1</code>')
      .replace(/\[([^\]]+)\]\((https?:\/\/[^\)]+)\)/g,'<a href="$2" target="_blank">$1</a>')
      .replace(/(^|[\s(])(https?:\/\/[^\s)<]+)/g,'$1<a href="$2" target="_blank">$2</a>')
      .replace(/\n/g,'<br>');
  }

  window.cfpSend=function(){
    if(busy) return;
    var key=$("gKey").value.trim(), inp=$("cfpInput"), txt=inp.value.trim();
    if(!txt||!key) return;
    inp.value=""; inp.style.height="auto";
    addMsg("user", txt.replace(/&/g,"&amp;").replace(/</g,"&lt;"));
    hist.push({role:"user",parts:[{text:txt}]});
    busy=true; $("cfpSendBtn").disabled=true; addTyping();

    fetch("https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key="+key,{
      method:"POST", headers:{"Content-Type":"application/json"},
      body:JSON.stringify({
        system_instruction:{parts:[{text:SYS}]},
        contents:hist,
        tools:[{google_search:{}}],
        generationConfig:{temperature:0.7,maxOutputTokens:4096}
      })
    })
    .then(function(r){ return r.ok ? r.json() : r.json().then(function(e){throw new Error(e.error?e.error.message:"API error")}); })
    .then(function(res){
      var el=$("cfpTyping"); if(el) el.remove();
      busy=false; $("cfpSendBtn").disabled=false;
      var txt="", src=[];
      if(res.candidates&&res.candidates[0]){
        var c=res.candidates[0];
        if(c.content&&c.content.parts) c.content.parts.forEach(function(p){if(p.text) txt+=p.text;});
        if(c.groundingMetadata&&c.groundingMetadata.groundingChunks)
          c.groundingMetadata.groundingChunks.forEach(function(ch){if(ch.web) src.push({uri:ch.web.uri,title:ch.web.title||""});});
      }
      if(!txt) txt="No response received. Please try again.";
      hist.push({role:"model",parts:[{text:txt}]});
      addMsg("bot",renderMd(txt),src);
    })
    .catch(function(err){
      var el=$("cfpTyping"); if(el) el.remove();
      busy=false; $("cfpSendBtn").disabled=false;
      addMsg("bot",'<strong style="color:#dc3545">Error:</strong> '+(err.message||err)+
        '<br><br>Check your Gemini API key. <a href="https://aistudio.google.com/apikey" target="_blank">Get a free key here</a>.');
    });
  };

  window.cfpAskTopic=function(kw){ $("cfpInput").value="Find open calls for papers related to: "+kw; cfpSend(); };
  window.cfpAskEx=function(btn){ $("cfpInput").value=(btn.textContent||btn.innerText).replace(/^[\s\S]?\s/,"").trim(); cfpSend(); };
})();
</script>
